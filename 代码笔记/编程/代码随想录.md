```
        return keep(nums,1);
```

### 3.3 比较含退格的元素
[844. 比较含退格的字符串 - 力扣（LeetCode）](https://leetcode.cn/problems/backspace-string-compare/description/)
由于需要退格，因此从后往前比较，遇到#就跳过。 然后一个一个进行比较。

遇到 #      → skip++
遇到普通字符 + skip>0 → 删除它，skip--
遇到普通字符 + skip=0 → 找到了有效字符

```
class Solution {
public:
    bool backspaceCompare(string s, string t) {
        int i = s.length()-1, j = t.length()-1;
        int skipS = 0, skipJ = 0; //保存有几个#

        while(i>=0 || j>=0){
            while(i>=0){ //找出满足条件的
                if(s[i] == '#'){
                    skipS +=1;
                    i--;
                }else if (s[i] !='#' && skipS >0){
                    skipS -=1;
                    i--;
                }else{
                    break;
                }
            }
            while(j>=0){
                if(t[j] == '#'){
                    skipJ +=1;
                    j--;
                }else if (t[j] !='#' && skipJ >0){
                    skipJ -=1;
                    j--;
                }else{
                    break;
                }
            }
            if(i>=0 && j>=0){ //处理完之后有可能出现空字符，因此需要先进行比较。
                if(s[i] != t[j]) return false;
            }else{ //进入else则说明至少有一个<0,也就是字符串结束了，此时如果i或者j还没有结束，说明长度不一样
                if(i>=0 or j>=0) return false; 
            }
            i--;
            j--; 

        }
        return true;
    }
};
```

##  4. 有序数组的平方
[977. 有序数组的平方 - 力扣（LeetCode）](https://leetcode.cn/problems/squares-of-a-sorted-array/submissions/743431494/)
注意这里的更新方式，for(初始化; 条件; 更新)， 然后由于要初始化声明两个变量，i和j，所以中间逗号，相当于是写了(初始化；条件；) 然后更新方式写在内部了。

```
class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        vector<int> result(nums.size(),0);
        int k =  nums.size()-1;

        for(int i = 0, j = nums.size()-1;i<=j;){
            if(nums[i]*nums[i]>=nums[j]*nums[j]){
                result[k--] = nums[i]*nums[i];
                i++;  
            }else{
                result[k--] = nums[j]*nums[j];
                j--;
            }
        }
        
        return result;
    }
};
```

```
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int i = 0;
        int sum = 0;
        int len = nums.size()+1;
        for(int j = 0;j<nums.size();j++){
                sum+=nums[j];
                
            while(sum>=target){
                len = min(len,j-i+1); #窗口合法，记录答案
                sum-=nums[i];
                i++;
            }
        }

        return len == nums.size()+1 ?0:len;
    }
    
};
```

## 6 螺旋矩阵
[54. 螺旋矩阵 - 力扣（LeetCode）](https://leetcode.cn/problems/spiral-matrix/)
这个题目要掌握边界判断条件，我的代码选择的是左闭右闭。
```
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int l = 0, r = matrix[0].size()-1, up = 0, down = matrix.size()-1;

        vector<int> ans;
        
        while(true){
            for(int i = l;i<=r;i++) ans.push_back(matrix[up][i]);
            if(++up>down) break;
            
//这里r由于需要是合法的区间，因此r的范围定义需要是matrix[0].size()-1,down同理。
//左闭右闭下左右相等合法，因此左>右说明遍历结束，跳出循环。

            for(int i = up;i<=down;i++) ans.push_back(matrix[i][r]);
            if(--r<l) break;

            for(int i = r;i>=l;i--) ans.push_back(matrix[down][i]);
            if(--down<up) break; 

            for(int i = down;i>=up;i--) ans.push_back(matrix[i][l]);
            if(++l>r) break;
        }

        return ans;
    }
};
```

```
class Solution {
```

```
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int, int> map;

        for(int i = 0;i<nums.size();i++){
            if(map.find(target-nums[i]) !=map.end()){ //表示在哈希表中找到了这个数字
```

```
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        int max_len = 0;
        unordered_set<long long> u_set;
        for(int i = 0;i<nums.size();i++){
            u_set.emplace(nums[i]);
        }

        for(auto& num:u_set){  //改成遍历一遍集合，来去除重复的
            if(u_set.find(num-1) == u_set.end()){ //如果没有找到num-1，那么num就可能是一个开头
                int len = 0;
                while(u_set.find(num + len)!= u_set.end()){
                    len++;
                    max_len = max(max_len,len);
                }
            }
        }
        return max_len;
    }
};
```

---
巧妙解法，把`result` 设置成 `Min`，**保证即使所有元素都是负数**，也能正确返回最大值。
举个例子，如果你的数组是：
`int nums[] = {-3, -5, -2, -8};`

那么最大子数组就是 `-2`，不能是 0。如果你一开始设 `result = 0`，那最后返回的值就会错。

 ✅ 为什么 `count` 从 0 开始就够了？
这是因为：

- `count` 表示的是“当前正在尝试扩展的子数组”的和；
- 如果这个和小于 0，就没有必要继续扩展，直接从下一个数重新开始子数组更优。
    
而从 0 开始，逐个加上数组元素就能模拟“从当前位置开始向后累加”的过程，一旦和为负就重新开始。
```
#define Min -100010
int maxSubArray(int* nums, int numsSize) {
    int result = Min; //巧妙
    int count = 0;  //巧妙
    for(int i =0;i<numsSize;i++){
        count+=nums[i];
        if(count>result) result = count;
        if(count<0) count = 0;
        
    }
    return result;
}
```

```
struct ListNode{

typedef struct ListNode ListNode;

    ListNode* dummyhead;
    int size;

List *create(){
    List *list = malloc(sizeof(List));
    ListNode *head = malloc(sizeof(ListNode));
    head->next = NULL;
    list->dummyhead = head;
    list->size = 0;
    return list;
}

void insert(List *list, int value){
    ListNode *node  = malloc(sizeof(ListNode));
    node->val = value;
    node->next = NULL;

    ListNode *tem = list->dummyhead;  //注意这里的处理，不要写出null指针

        tem = tem->next;
    }
    tem->next = node;
```

- `children` 是一个 **指向 Node 指针的数组**。  也就是children里存放了一堆node指针
- `children[i]` 是一个 **第 i 个子节点**，它的类型是 `struct Node*`

因此`root->children[i]`的意思也就是相当于`root->left`这种，类比一下，都是指针指向指针。也相当于是后序遍历，遍历到最后了之后进行+1的操作。逐层返回上来。

[222. 完全二叉树的节点个数 - 力扣（LeetCode）](https://leetcode.cn/problems/count-complete-tree-nodes/) 对于这个题目， 可以使用到二叉树的特性，当树是完全二叉树的时候，2^n -1 即为节点的个数。其中n是深度。深度从根节点开始为1.  
```
typedef struct TreeNode TreeNode;
int countNodes(struct TreeNode* root) {
    if(root ==NULL) return 0;
    TreeNode *left = root->left, *right = root->right;
    int leftCount = 0, rightCount = 0;
    while(left){
        left = left->left;
        leftCount++;
    }
    while(right){
        right = right->right;
        rightCount++;
    }
    if(leftCount ==rightCount) return (2<<leftCount)-1;  //相当于2的深度次方-1，满二叉树特性

    int leftTree = countNodes(root->left);
    int rightTree = countNodes(root->right);//这里不能写left，它和上面定义的不一样，因为left已经被while循环不断递归掉了
    int result = leftTree + rightTree +1;
    return result;
}
```

由于我定义left的时候，已经遍历 了一次，但是`leftCount`设置为从0开始，也就是比如深度为2的时候，`leftCount`经过++才是1。 这里有一个位运算的操作逻辑，`2<< leftCount` 当`leftCount`是1的时候相当于是2的2次方，也就是正好和深度对应了。 当然这个题目也可以使用经典的递归逻辑，后序遍历即可，适用于所有二叉树。
```
int countNodes(struct TreeNode* root) {
    if(root ==NULL) return 0;
    int left = countNodes(root->left);
    int right = countNodes(root->right);
    int result = left + right +1;
    return result;
}
```

# 6. 回溯算法

回溯算法适合解决诸如排列组合，棋盘等解法，本质是暴力解法。

[77. 组合 - 力扣（LeetCode）](https://leetcode.cn/problems/combinations/) 对于这个组合问题，需要注意是在path上不断进行修改，共享同一个path内存， 需要使用深拷贝，否则最终得到的结果都会变成最后一个组合的结果，也就是先前的结果由于地址改变了也被同步修改了
```
int *path, pathTop;
int **ans, ansTop;

void backtracking(int n, int k, int startIndex) {
    if (pathTop == k){
        int *tem = malloc(k*sizeof(int));
        for(int i = 0;i<k;i++){
            tem[i] = path[i]; //这里是一个深拷贝逻辑，防止path因后续改变而导致先前保存结果同步改变
        }
        ans[ansTop++] = tem;
        return;
    }
    for(int j = startIndex;j<=n-(k-pathTop)+1;j++){  //这里进行了剪枝的操作，逻辑就是找到最晚开始的节点。
        path[pathTop++] = j;
        backtracking(n,k,j+1);
        pathTop--;//移动栈顶指针即可，等待后续把它覆盖了
    }
}

int** combine(int n, int k, int* returnSize, int** returnColumnSizes) {
    path = malloc(k*sizeof(int));
```

```
            tem[i] = path[i];

        sum -= candidates[i];
        pathTop--;
    }
}
```

```C
int** generateMatrix(int n, int* returnSize, int** returnColumnSizes) {
    int **ans = malloc(n*sizeof(int *));
    *returnSize = n;
    *returnColumnSizes = malloc(n*sizeof(int));
    for(int i = 0;i<n;i++){
        ans[i] = malloc(n*sizeof(int));
        (*returnColumnSizes)[i] = n;
    }

    int left = 0, right = n-1, top = 0, down = n-1;
    int num = 1;
    while(left<=right && top<=down){
        for(int i = left;i<=right;i++) ans[top][i] = num++;
        top++;
        for(int i = top;i<=down;i++) ans[i][right] = num++;
        right--;
        for(int i = right;i>=left;i--) ans[down][i] = num++;
        down--;
        for(int i = down;i>=top;i--) ans[i][left] = num++;
        left++;
    }
    return ans; 
}
```
