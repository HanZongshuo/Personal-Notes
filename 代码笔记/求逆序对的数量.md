```C
#include <stdio.h>

#define N 100010

typedef long long LL;

int n;
int q[N]; // 原数组（从1开始）
int t[N]; // 临时数组
LL res = 0; // 逆序对个数

void merge_sort(int q[], int l, int r) {
    if (l >= r) return;

    int mid = (l + r) / 2;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);

    int i = l, j = mid + 1, k = 0;

    while (i <= mid && j <= r) {
        if (q[i] <= q[j]) {
            t[k++] = q[i++];
        } else {
            t[k++] = q[j++];
            res += (LL)(mid - i + 1); // 注意类型转为 long long
        }
    }

    while (i <= mid) t[k++] = q[i++];
    while (j <= r) t[k++] = q[j++];

    for (i = l, j = 0; i <= r; i++, j++) {
        q[i] = t[j];
    }
}

int main() {
    printf("Enter number of elements:\n");
    scanf("%d", &n);
    printf("Enter the numbers:\n");
    for (int i = 1; i <= n; i++) {
        scanf("%d", &q[i]);
    }

    merge_sort(q, 1, n);

    printf("Number of inversions: %lld\n", res);

    return 0;
}

```

归并排序会不断把数组拆分成左右两个**有序子数组**，然后合并。

我们可以利用这个合并的过程高效地统计逆序对。
