```C
#include<stdio.h>
#include<string.h>

int main(){
    char str[1000];
    
    printf("Input the text separate by space.\n");
    fgets(str,sizeof(str),stdin);

    int i = 0;
    int n = strlen(str);

    while(i<n){
        while (i < n && str[i] == ' ') {
            i++;  // 跳过空格
        }

        int j = i;

        while (j<n && str[j] != ' '){
            j++;
        }

        for (int k = i; k<j +1; k++){
            printf("%c",str[k]);
        }

        printf("\n");

        i = j;
    
    } 
    return 0;
}
```
