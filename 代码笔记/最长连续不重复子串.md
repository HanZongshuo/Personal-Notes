这里的连续指的是，下标连续即可，无需数组里面具体的值连续

```C
#include<stdio.h>
#define N 10000

int main(){
    int a[N];
    int s[N];
    int res = 0;

 // 这里直接输入 1 2 2 3 5也是可以的
    printf("Input the numbers:\n");
    for(int i =1; i<6;i++){
        scanf("%d",&a[i]);
    }

    int j = 1;
    for(int i = 1; i<6; i++){
        s[a[i]]++;

        while (s[a[i]]>1){
            s[a[j]] --;
            j++;
        }
        if(res< i-j +1){
            res = i-j +1;
        }
        
    }
    printf("Result is :%d\n",res);

    return 0;

}
```
