```C
#include<stdio.h>

int lowBit(int x){
        return x&(-x);
}

int main(){
    int n;
    printf("How many numbers you want to input:\n");
    scanf("%d",&n);
    printf("Input the numbers.");
    while (n--)
    {
        int x, res = 0;
        scanf("%d",&x);
        while (x)
        {
            x = x-lowBit(x);
            res++;
        }
        printf("%d ",res);
        
    }    
    return 0;
}
```
