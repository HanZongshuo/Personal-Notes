分配两次的原因在于每一次字符串长度不一致，或者不清楚

```C
#include<stdio.h>
#include<stdlib.h>

char **generateStrings(int n){
    char **str = (char**)malloc(n*sizeof(char*));
    if(str ==NULL){
        printf("Allocate memory failed.\n");
        return NULL;
    }

    for(int i = 0;i<n;i++){
        str[i] = (char*)malloc(10*sizeof(char));
        if(str[i]==NULL){
            printf("Allocate memory failed.\n");
            return NULL;
        }
        printf("Enter string %d:\n",i+1);
        scanf(" %s",str[i]);
    }
    return str;
}
int main(){
    printf("Enter the number of string:\n");
    int n;
    scanf("%d",&n);
    char ** str = generateStrings(n);
    printf("The stored strings are:\n");
    for(int i=0;i<n;i++){
        printf("%d: %s\n",i+1,str[i]);
        free(str[i]);
    }
    free(str);
    
}
```
