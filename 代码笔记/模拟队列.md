```C
#include<stdio.h>
#include<string.h>
#define N 100010

int  hh= 0, tt = -1, stk[N];
//这里的head值大于-1， 代表还没有初始化

void push( int x){
    stk[++tt] = x;
}

void pop(){
    hh++;
}

void empty(){
    if (tt >= hh) printf("NO\n");
    else printf("YES\n");
}

void query(){
    printf("%d\n",stk[hh]);
}

int main(){
    int m;
    scanf("%d",&m);
    while (m--)
    {
        char op[6]; int x;
        scanf(" %s",op);
        if(strcmp(op,"push") == 0){
            scanf(" %d",&x);push(x);
        }else if(strcmp(op,"query") == 0){
            query();
        }else if(strcmp(op,"pop") == 0){
            pop();
        }else if(strcmp(op,"empty") == 0){
            empty();
        }
    }
    return 0;
}
```
