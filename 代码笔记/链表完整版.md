#include<stdio.h>
#include<stdlib.h>

typedef struct node{
    int value;
    struct node* next;
}Node;

typedef struct{
    Node *head;
}List;

List *create(){
    List *list = malloc(sizeof(List));
    list->head = NULL;
    return list;
}

void insert_head(List *l,int value){
    Node *n = malloc(sizeof(Node));
    n->value = value;
    n->next = l->head;
    l->head = n;
}

void insert_back(List *l, int value){
    Node *n = malloc(sizeof(Node));
    n->value = value;

    if(l->head==NULL){
        l->head = n;
    }else{
        Node *cur = l->head;
        while (cur->next !=NULL)
        {
            cur = cur->next;
        }
        cur->next = n;
        
    }
}

void print_list(Node *ptr){
    int i = 0;
    while (ptr!=NULL)
    {
        printf("The %d node value is %d\n",i+1,ptr->value);
        i++;
        ptr = ptr->next;
    } 
}

void reverse(List* l){
    Node *pre = NULL;
    Node *cur = l->head;
    Node *next = NULL;

    while (cur!=NULL)
    {
        next = cur->next;
        cur->next = pre;
        pre = cur;
        cur = next;
    }
    l->head = pre;

}

int main(){
    List *list = create();
    int nums[10];
    int i = 0;

    printf("Input the value:\n");
    while (scanf(" %d",&nums[i])==1)
    {
        insert_back(list,nums[i]);
        i++;
    }
    
    print_list(list->head);
    printf("\n");

    reverse(list);
    print_list(list->head);
     
}
